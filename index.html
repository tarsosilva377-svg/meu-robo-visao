// NAVIO LEVE PRO JOGO - copia essa função pro seu jogo function createShipLeve(cor){ const g = new THREE.Group(); // Materiais sem sombra, sem brilho pesado const cinza = new THREE.MeshStandardMaterial({color:0x8a8f96, roughness:0.7, metalness:0.2}); const escuro = new THREE.MeshStandardMaterial({color:0x23262b, roughness:0.7, metalness:0.2}); const preto = new THREE.MeshStandardMaterial({color:0x111111, roughness:0.7}); const vermelho = new THREE.MeshStandardMaterial({color:0xb02a1e, roughness:0.7}); // CASCO - só 1 forma, sem bevel pesado const s = new THREE.Shape(); s.moveTo(0,-7); s.lineTo(1.8,-6.5); s.lineTo(2,-2); s.lineTo(1.3,3); s.lineTo(0,7.5); s.lineTo(-1.3,3); s.lineTo(-2,-2); s.lineTo(-1.8,-6.5); s.closePath(); const casco = new THREE.Mesh(new THREE.ExtrudeGeometry(s,{depth:1.2, bevelEnabled:false}), cinza); casco.rotation.x = Math.PI/2; g.add(casco); // CONVÉS - 1 caixa só const deck = new THREE.Mesh(new THREE.BoxGeometry(3,0.2,12), new THREE.MeshStandardMaterial({color:cor})); deck.position.y = 0.7; g.add(deck); // SUPERESTRUTURA - 2 caixas, sem Extrude const sup = new THREE.Mesh(new THREE.BoxGeometry(2,1.2,4), cinza); sup.position.set(0,1.4,-1); sup.rotation.x = -0.05; g.add(sup); const ponte = new THREE.Mesh(new THREE.BoxGeometry(1.6,0.9,1.8), cinza); ponte.position.set(0,2.3,-0.2); g.add(ponte); // Janela const win = new THREE.Mesh(new THREE.BoxGeometry(1.5,0.2,0.1), preto); win.position.set(0,2.5,0.75); g.add(win); // Mastro - 1 cilindro só const mast = new THREE.Mesh(new THREE.CylinderGeometry(0.08,0.15,1.5,6), escuro); mast.position.set(0,3.4,-1.2); g.add(mast); // Canhão const tur = new THREE.Mesh(new THREE.CylinderGeometry(0.4,0.55,0.4,6), cinza); tur.position.set(0,0.9,4.2); g.add(tur); const cano = new THREE.Mesh(new THREE.CylinderGeometry(0.1,0.12,1.8,6), preto); cano.rotation.x = Math.PI/2; cano.position.set(0,1,5.2); g.add(cano); // Heliponto const heli = new THREE.Mesh(new THREE.CylinderGeometry(0.85,0.85,0.1,10), vermelho); heli.position.set(0,0.85,-4.8); g.add(heli); return g; }
